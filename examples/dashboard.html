-- Revenue trend: invoiced sales vs cash collected, per calendar month.
CREATE VIEW kpi_monthly_revenue AS
WITH sales AS (
    SELECT month_key,
           COUNT(*)                          AS orders,
           COUNT(DISTINCT customer_id)       AS active_customers,
           SUM(order_revenue)                AS revenue,
           SUM(order_revenue - order_cost)   AS gross_margin
    FROM v_orders
    WHERE status = 'Completed'
    GROUP BY month_key
), cash AS (
    SELECT strftime('%Y-%m', paid_on) AS month_key, SUM(amount) AS collections
    FROM payments
    GROUP BY strftime('%Y-%m', paid_on)
)
SELECT
    s.month_key,
    s.orders,
    s.active_customers,
    ROUND(s.revenue, 2)                          AS revenue,
    ROUND(s.gross_margin, 2)                     AS gross_margin,
    ROUND(s.revenue / NULLIF(s.orders, 0), 2)    AS avg_order_value,
    ROUND(COALESCE(c.collections, 0), 2)         AS collections
FROM sales s
LEFT JOIN cash c ON c.month_key = s.month_key;

Show SQL · kpi_region_sales
-- Sales by region with each region's share of total revenue.
CREATE VIEW kpi_region_sales AS
SELECT
    region_name,
    COUNT(DISTINCT customer_id)               AS active_customers,
    COUNT(*)                                  AS orders,
    ROUND(SUM(order_revenue), 2)              AS revenue,
    ROUND(SUM(order_revenue - order_cost), 2) AS gross_margin,
    ROUND(100.0 * SUM(order_revenue) / SUM(SUM(order_revenue)) OVER (), 1) AS revenue_share_pct
FROM v_orders
WHERE status = 'Completed'
GROUP BY region_name;

-- Units, revenue and margin per product, ranked by revenue.
CREATE VIEW kpi_product_sales AS
SELECT
    l.product_id,
    l.product_name,
    l.category,
    SUM(l.quantity)                           AS units,
    ROUND(SUM(l.line_revenue), 2)             AS revenue,
    ROUND(SUM(l.line_revenue - l.line_cost), 2) AS gross_margin,
    RANK() OVER (ORDER BY SUM(l.line_revenue) DESC) AS revenue_rank
FROM v_order_lines l
JOIN orders o ON o.order_id = l.order_id
WHERE o.status = 'Completed'
GROUP BY l.product_id, l.product_name, l.category;

Receivables aging

Unpaid invoices by days past due date, as of 2025-12-31 · region and segment filters apply

Show SQL · v_open_invoices + kpi_ar_aging
-- Unpaid invoices with their aging bucket (days past due date).
CREATE VIEW v_open_invoices AS
SELECT
    b.*,
    CASE WHEN b.days_past_due <= 0  THEN 1
         WHEN b.days_past_due <= 30 THEN 2
         WHEN b.days_past_due <= 60 THEN 3
         WHEN b.days_past_due <= 90 THEN 4
         ELSE 5 END AS bucket_order,
    CASE WHEN b.days_past_due <= 0  THEN 'Not yet due'
         WHEN b.days_past_due <= 30 THEN '1-30 days'
         WHEN b.days_past_due <= 60 THEN '31-60 days'
         WHEN b.days_past_due <= 90 THEN '61-90 days'
         ELSE '90+ days' END AS aging_bucket
FROM v_invoice_balances b
WHERE b.outstanding > 0.005;

-- Receivables aging: how much customers owe, by how late it is.
CREATE VIEW kpi_ar_aging AS
SELECT
    bucket_order,
    aging_bucket,
    COUNT(*)                  AS open_invoices,
    ROUND(SUM(outstanding), 2) AS outstanding,
    ROUND(100.0 * SUM(outstanding) / SUM(SUM(outstanding)) OVER (), 1) AS pct_of_total
FROM v_open_invoices
GROUP BY bucket_order, aging_bucket;

Customer retention by first-order month

% of each cohort that ordered again N months later · region and segment filters apply

Show SQL · kpi_cohort_retention
-- Customers grouped by the month of their first order; what share of each
-- group ordered again N months later.
CREATE VIEW kpi_cohort_retention AS
WITH completed AS (
    SELECT customer_id, order_date FROM v_orders WHERE status = 'Completed'
), first_order AS (
    SELECT customer_id, MIN(order_date) AS first_date
    FROM completed
    GROUP BY customer_id
), activity AS (
    SELECT DISTINCT
        f.customer_id,
        strftime('%Y-%m', f.first_date) AS cohort_month,
        (CAST(strftime('%Y', c.order_date) AS INTEGER) * 12 + CAST(strftime('%m', c.order_date) AS INTEGER)) - (CAST(strftime('%Y', f.first_date) AS INTEGER) * 12 + CAST(strftime('%m', f.first_date) AS INTEGER)) AS months_since_first
    FROM first_order f
    JOIN completed c ON c.customer_id = f.customer_id
), sizes AS (
    SELECT cohort_month, COUNT(*) AS cohort_size
    FROM activity
    WHERE months_since_first = 0
    GROUP BY cohort_month
)
SELECT
    a.cohort_month,
    a.months_since_first,
    s.cohort_size,
    COUNT(*) AS active_customers,
    ROUND(100.0 * COUNT(*) / s.cohort_size, 1) AS retention_pct
FROM activity a
JOIN sizes s ON s.cohort_month = a.cohort_month
GROUP BY a.cohort_month, a.months_since_first, s.cohort_size;

Data-quality checks

Run on the raw tables before any KPI is trusted · not affected by filters

Show SQL · dq_issues
CREATE VIEW dq_issues AS
SELECT 'order_without_items' AS check_name, 'error' AS severity, 'orders' AS table_name,
       o.order_id AS record_id, 'Order has no line items' AS detail
FROM orders o
WHERE NOT EXISTS (SELECT 1 FROM order_items i WHERE i.order_id = o.order_id)

UNION ALL
SELECT 'orphan_order_customer', 'error', 'orders', o.order_id, 'customer_id not found in customers'
FROM orders o
LEFT JOIN customers c ON c.customer_id = o.customer_id
WHERE c.customer_id IS NULL

UNION ALL
SELECT 'orphan_item_product', 'error', 'order_items', i.order_item_id, 'product_id not found in products'
FROM order_items i
LEFT JOIN products p ON p.product_id = i.product_id
WHERE p.product_id IS NULL

UNION ALL
SELECT 'bad_quantity_or_price', 'error', 'order_items', i.order_item_id,
       'quantity <= 0, negative price or discount outside 0-100'
FROM order_items i
WHERE i.quantity <= 0 OR i.unit_price < 0 OR i.discount_pct < 0 OR i.discount_pct > 100

UNION ALL
SELECT 'due_before_order', 'error', 'orders', o.order_id, 'due_date is before order_date'
FROM orders o
WHERE o.due_date < o.order_date

UNION ALL
SELECT 'orphan_payment', 'error', 'payments', p.payment_id, 'order_id not found in orders'
FROM payments p
LEFT JOIN orders o ON o.order_id = p.order_id
WHERE o.order_id IS NULL

UNION ALL
SELECT 'payment_before_order', 'error', 'payments', p.payment_id, 'paid_on is before the order_date'
FROM payments p
JOIN orders o ON o.order_id = p.order_id
WHERE p.paid_on < o.order_date

UNION ALL
SELECT 'payment_on_cancelled_order', 'warning', 'payments', p.payment_id, 'payment recorded against a cancelled order'
FROM payments p
JOIN orders o ON o.order_id = p.order_id
WHERE o.status = 'Cancelled'

UNION ALL
SELECT 'overpaid_invoice', 'warning', 'orders', b.order_id, 'payments exceed the invoice amount'
FROM v_invoice_balances b
WHERE b.outstanding < -0.005

UNION ALL
SELECT 'duplicate_customer_email', 'warning', 'customers', c.customer_id, 'email already used by an older customer record'
FROM customers c
WHERE EXISTS (
    SELECT 1 FROM customers c2
    WHERE LOWER(c2.email) = LOWER(c.email) AND c2.customer_id < c.customer_id
)

UNION ALL
SELECT 'missing_customer_region', 'warning', 'customers', c.customer_id, 'region_id is empty or unknown'
FROM customers c
LEFT JOIN regions r ON r.region_id = c.region_id
WHERE r.region_id IS NULL

UNION ALL
SELECT 'missing_email', 'warning', 'customers', c.customer_id, 'email is empty'
FROM customers c
WHERE c.email IS NULL OR c.email = '';
